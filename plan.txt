trace_ray(ray):
    // simplified version of taking a ray and getting the first primitive hit
    hit = intersect(ray)
    if !hit:
        return bkg_color
    
    // recursively trace reflected rays
    
    if (hit.is_glossy)
        reflected_ray = reflect(ray, hit)
        reflected_color = trace_ray(reflected_ray)
    
    if (hit.is_transparent)
        refracted_ray = refract(ray, hit)
        refracted_color = trace_ray(refracted_ray)
